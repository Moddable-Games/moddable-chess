Game Controller SDK Demo

Everything createGameController can do, demonstrated live

Variant Gallery

Multiple variants rendered simultaneously. Each controller is independent with its own game state, AI, and event listeners.

View source 
import MCE from './js/chess-engine.js';
import './js/chess-moves.js';
import './js/chess-play.js';
import './js/chess-variants.js';
import './js/board-renderer.js';
import './js/chess-ai.js';
import { createGameController } from './js/game-controller-core.js';

const VARIANTS = ['standard', 'atomic', 'kingOfTheHill', 'capablanca', 'horde', 'racingKings'];

VARIANTS.forEach(key => {
  const game = MCE.createGame(key);
  const container = document.createElement('div');
  gallery.appendChild(container);

  createGameController(container, game, {
    players: { w: 'ai', b: 'ai' },
    aiDepth: 3,
    onGameEnd: (result) => console.log(key, result)
  });
});

Programmatic Control

Load positions from FEN, make moves, undo, trigger AI, and observe the event stream in real time.

View source 
import MCE from './js/chess-engine.js';
import { createGameController } from './js/game-controller-core.js';
// ... other imports omitted for brevity

const game = MCE.createGame('standard');
const ctrl = createGameController(
  document.getElementById('board'), game, {
    players: { w: 'human', b: 'human' },
    onMove: (move) => log(`Move: ${MCE.sqToAlgebraic(move.from, game)}${MCE.sqToAlgebraic(move.to, game)}`),
    onGameEnd: (result) => log(`Game over: ${result}`),
    onTurnChange: (turn) => log(`Turn: ${turn}`)
  }
);

// Programmatic API
ctrl.undo();
ctrl.setFlipped(true);
ctrl.setDifficulty(4);
ctrl.getGame(); // current game object

Render Hooks

The controller accepts custom rendering functions for terrain painting, piece replacement, post-render overlays, and surround decorations.

View source 
import MCE from './js/chess-engine.js';
import { createGameController } from './js/game-controller-core.js';

const game = MCE.createGame('standard');
const ctrl = createGameController(container, game, {
  players: { w: 'human', b: 'ai' },
  renderOpts: {
    // Replace default square colours
    tilePainter: function(sq, row, col, isLight) {
      if (game.terrain && game.terrain[sq] === 'water') return '#2196F3';
      return null; // use default
    },

    // Replace piece rendering (e.g. custom SVG sprites)
    pieceProvider: function(piece, sq, tileSize) {
      const img = document.createElementNS('http://www.w3.org/2000/svg', 'image');
      img.setAttribute('href', `assets/pieces/${piece}.svg`);
      img.setAttribute('width', tileSize);
      img.setAttribute('height', tileSize);
      return img;
    },

    // Draw after the board renders (overlays, arrows, labels)
    afterRender: function(svg, game, opts) {
      const from = MCE.sq(6, 4, game), to = MCE.sq(4, 4, game);
      drawArrow(svg, from, to, opts);
    },

    // Draw around the board (coordinate labels, borders)
    surroundRenderer: function(svg, boardRect, opts) {
      addCoordinateLabels(svg, boardRect, opts);
    }
  }
});

AI Configuration

Five difficulty presets with configurable search depth. The AI runs in a Web Worker to keep the UI responsive.

View source 
// Difficulty presets
MCE.AI_DIFFICULTIES
// { beginner: { depth: 1 }, easy: { depth: 2 }, medium: { depth: 3 }, hard: { depth: 4 }, expert: { depth: 5 } }

// Use with controller
const ctrl = createGameController(container, game, {
  players: { w: 'human', b: 'ai' },
  aiDepth: 4,
  onMove: (move, undo, captured) => {
    if (captured) playSound('capture');
    else playSound('move');
  }
});

// Change difficulty mid-game
ctrl.setDifficulty(5); // expert

// Or use aiPickMove directly for headless contexts (Node.js, Workers)
const bestMove = MCE.aiPickMove(game, 4, { variant: 'atomic' });
